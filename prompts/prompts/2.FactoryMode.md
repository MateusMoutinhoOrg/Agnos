### Objective:
- aply the factory style for all objects of these project
   - aply the factory style in sandbox
   - aply the factory style on adapters

- create a spec for factory and a rule that all objects must be created using the factory mode.
Factory style :

~~~go 
package main

type Car struct {
	name      string
	working   bool
	TurnOn    func()
	ShowProps func()
}

func TurnOnFactory(car *Car) func() {
	return func() {
		car.working = true
	}
}
func ShowPropsFactory(car *Car) func() {
	return func() {
		println(car.name)
		println(car.working)
	}
}
func NewCar(name string) *Car {
	var selfOjbect Car
	selfOjbect.name = name
	selfOjbect.TurnOn = TurnOnFactory(&selfOjbect)
	selfOjbect.ShowProps = ShowPropsFactory(&selfOjbect)
	return &selfOjbect
}
func main() {
	uno := NewCar("uno")
	cargo := NewCar("cargo")

	uno.TurnOn()
	cargo.TurnOn()
	uno.ShowProps()

	cargo.ShowProps()
}
~~~
